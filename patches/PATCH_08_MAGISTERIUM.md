# PATCH 08 — Magisterium AI Commentary

## After this patch the user can

Select a verse, switch to "Ask AI", type a question, and receive an answer grounded exclusively in Catholic Magisterial documents — with the source documents cited beneath every response.

## Prerequisites

- Patch 07 complete
- User has a Magisterium AI account at [magisteriummagi.ai](https://magisteriummagi.ai) and a developer API key
- `MAI_API_KEY` and `MAI_API_URL` set in `config.js`

## Critical constraint

**This tab must never call any AI service other than Magisterium AI.** If their API is unreachable, show a link to their website — do not fall back to another service.

## Files to create

- `js/commentary.js`

## Files to modify

- `index.html` — replace AI tab pane content
- `js/app.js` — wire up commentary on tab switch

2. The authentication header format (Bearer token? API key header? Confirm from docs)
3. The request body shape (messages array? single prompt? model name?)
4. Whether responses include citations/sources in the response body

**Update `config.example.js` with whatever the actual URL and auth format are.**

The implementation below uses a plausible REST shape that mirrors common LLM APIs. Adjust the request body and response parsing to match Magisterium AI's actual schema.

## js/commentary.js

```js
import { CONFIG } from '../config.js';
import { state } from './app.js';

// ── Public entry point ────────────────────────────────────

export function initCommentary() {
  const btn = document.getElementById('aiAskBtn');
  if (!btn) return;
  btn.addEventListener('click', handleAsk);

  // Allow Ctrl+Enter / Cmd+Enter to submit
  document.getElementById('aiQuestion')?.addEventListener('keydown', e => {
    if (e.key === 'Enter' && (e.ctrlKey || e.metaKey)) handleAsk();
  });
}

export function setVerseContext(verseId, verseText) {
  const el = document.getElementById('aiVerseContext');
  if (!el) return;
  if (!verseId) {
    el.textContent = 'No verse selected.';
    return;
  }
  el.innerHTML = `<strong>${formatVerseId(verseId)}</strong><br>"${verseText || '…'}"`;
}

// ── Ask handler ───────────────────────────────────────────

async function handleAsk() {
  const question = document.getElementById('aiQuestion')?.value?.trim();
  if (!question) return;

  if (!CONFIG.MAI_API_KEY || CONFIG.MAI_API_KEY === 'GET_FROM_MAGISTERIUMMAGI.AI') {
    showError('Magisterium AI API key not configured. Add your key to config.js.');
    return;
  }

  if (!state.selectedVerseId) {
    showError('Please select a verse first by clicking on it in the text.');
    return;
  }

  showLoading();
  document.getElementById('aiAskBtn').disabled = true;

  try {
    const response = await callMagisteriumAI(question);
    renderResponse(response);
  } catch (err) {
    showError(`Could not reach Magisterium AI.<br>
      <a href="https://magisteriummagi.ai" target="_blank" class="ref-link">
        Use Magisterium AI directly ↗
      </a><br>
      <small>${err.message}</small>`);
  } finally {
    document.getElementById('aiAskBtn').disabled = false;
  }
}

// ── API call ──────────────────────────────────────────────

async function callMagisteriumAI(question) {
  const verseRef  = formatVerseId(state.selectedVerseId);
  const verseText = state.selectedVerseText || '';

  // Construct a context-rich prompt
  const userMessage = [
    `Biblical verse: ${verseRef}`,
    verseText ? `"${verseText}"` : '',
    '',
    `Question: ${question}`,
    '',
    'Please answer based only on Catholic Magisterial teaching and Sacred Scripture, and cite your sources.'
  ].filter(Boolean).join('\n');

  // ⚠️ VERIFY THIS AGAINST MAGISTERIUM AI'S ACTUAL DOCS BEFORE SHIPPING
  // The shape below mirrors common LLM chat APIs. Adjust as needed.
  const res = await fetch(`${CONFIG.MAI_API_URL}chat/completions`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${CONFIG.MAI_API_KEY}`,
      // Some APIs use a different header: 'X-API-Key', 'api-key', etc.
      // Check Magisterium AI docs and update accordingly.
    },
    body: JSON.stringify({
      model:      'magisterium-1',  // Verify model name from their docs
      messages:   [{ role: 'user', content: userMessage }],
      max_tokens: 1000,
    }),
  });

  if (!res.ok) {
    const errBody = await res.text();
    throw new Error(`API returned ${res.status}: ${errBody.slice(0, 200)}`);
  }

  return res.json();
}

// ── Response rendering ────────────────────────────────────

function renderResponse(data) {
  const responseEl = document.getElementById('aiResponse');
  if (!responseEl) return;

  // Parse the response — adjust based on actual Magisterium AI response shape
  // Common shapes: data.choices[0].message.content (OpenAI-compatible)
  //                data.answer + data.sources (custom shape)
  //                data.content[0].text (Anthropic-style)

  let answerText = '';
  let sources    = [];

  if (data.choices?.[0]?.message?.content) {
    // OpenAI-compatible response
    answerText = data.choices[0].message.content;
    sources    = data.choices[0].sources || [];
  } else if (data.answer) {
    // Possible custom Magisterium AI shape
    answerText = data.answer;
    sources    = data.sources || data.citations || [];
  } else if (data.content?.[0]?.text) {
    // Anthropic-style
    answerText = data.content[0].text;
    sources    = data.sources || [];
  } else {
    // Fallback: stringify whatever we got
    answerText = JSON.stringify(data, null, 2);
    console.warn('Unexpected Magisterium AI response shape. Update renderResponse() in commentary.js.');
  }

  const sourcesHtml = sources.length > 0 ? `
    <div class="ai-sources">
      <div class="ai-sources__label">Sources cited</div>
      ${sources.map(s => `
        <div class="ai-source-item">${typeof s === 'string' ? s : (s.title || s.name || JSON.stringify(s))}</div>
      `).join('')}
    </div>
  ` : '<p style="font-size:0.8rem;color:var(--color-text-muted);margin-top:0.5rem">No explicit sources returned. Check magisteriummagi.ai for source details.</p>';

  responseEl.style.display = 'block';
  responseEl.innerHTML = `
    <div class="ai-answer-text">${answerText.replace(/\n/g, '<br>')}</div>
    ${sourcesHtml}
  `;
}

// ── Loading and error states ──────────────────────────────

function showLoading() {
  const el = document.getElementById('aiResponse');
  if (!el) return;
  el.style.display = 'block';
  el.innerHTML = `<div class="ai-loading">
    <div class="ai-spinner"></div>
    Consulting Magisterium AI…
  </div>`;
}

function showError(msg) {
  const el = document.getElementById('aiResponse');
  if (!el) return;
  el.style.display = 'block';
  el.innerHTML = `<div class="error-msg">${msg}</div>`;
}

// ── Helpers ───────────────────────────────────────────────

function formatVerseId(verseId) {
  if (!verseId) return '';
  const [book, ch, v] = verseId.split('.');
  return `${book} ${ch}:${v}`;
}
```
